// Fixed issuer/key endpoint; RS256 verification uses platform WebCrypto, never a client-selected key URL.
const encoder=new TextEncoder();let keyCache=null,keyUntil=0;
function decode(part){return Uint8Array.from(atob(part.replace(/-/g,'+').replace(/_/g,'/')),c=>c.charCodeAt(0));}
export async function verifyGoogleToken(token,clientId,nonce,fetchKeys=()=>fetch('https://www.googleapis.com/oauth2/v3/certs')){
 if(typeof token!=='string'||token.length>12000||!clientId||!nonce)throw Error('Invalid sign-in');
 const parts=token.split('.');if(parts.length!==3)throw Error('Invalid token');
 const head=JSON.parse(new TextDecoder().decode(decode(parts[0])));
 if(head.alg!=='RS256'||typeof head.kid!=='string')throw Error('Invalid signing algorithm');
 // Tests supply their own fetcher, but production always uses the fixed Google endpoint.
 if(!keyCache||Date.now()>keyUntil||!keyCache.some(k=>k.kid===head.kid)){
  const response=await fetchKeys();if(!response.ok)throw Error('Google verification unavailable');
  const body=await response.json();if(!Array.isArray(body.keys))throw Error('Invalid keys');
  keyCache=body.keys;keyUntil=Date.now()+60*60*1000;
 }
 const jwk=keyCache.find(k=>k.kid===head.kid&&k.kty==='RSA'&&(!k.use||k.use==='sig')&&(!k.alg||k.alg==='RS256'));
 if(!jwk)throw Error('Unknown key');
 const key=await crypto.subtle.importKey('jwk',jwk,{name:'RSASSA-PKCS1-v1_5',hash:'SHA-256'},false,['verify']);
 if(!await crypto.subtle.verify('RSASSA-PKCS1-v1_5',key,decode(parts[2]),encoder.encode(parts[0]+'.'+parts[1])))throw Error('Invalid signature');
 const claims=JSON.parse(new TextDecoder().decode(decode(parts[1]))),now=Math.floor(Date.now()/1000);
 if(!['accounts.google.com','https://accounts.google.com'].includes(claims.iss)||claims.aud!==clientId||
  (claims.azp&&claims.azp!==clientId)||typeof claims.exp!=='number'||claims.exp<=now||
  typeof claims.iat!=='number'||claims.iat>now+60||claims.iat<now-7200||
  (claims.nbf&&claims.nbf>now+60)||typeof claims.sub!=='string'||!claims.sub||claims.sub.length>255||claims.nonce!==nonce)throw Error('Invalid token claims');
 return claims;
}

const bad=message=>{throw Object.assign(Error(message),{status:400});};
function date(value){
 if(!value)return '';
 if(!/^\d{4}-\d{2}-\d{2}$/.test(value)||!Number.isFinite(Date.parse(value+'T00:00:00Z'))||new Date(value+'T00:00:00Z').toISOString().slice(0,10)!==value)bad('Use valid dates in YYYY-MM-DD format.');
 return value;
}
export async function ownerPeople(env,p,ownerId){
 const start=date(p.get('start')),end=date(p.get('end'));
 if(start&&end&&start>end)bad('Start date must be on or before end date.');
 const mode=p.get('mode')||'',kind=p.get('kind')||'';
 if(mode&&!['vfr','ifr','unknown'].includes(mode))bad('Invalid training mode.');
 if(kind&&!['guest','member'].includes(kind))bad('Invalid user type.');
 const offset=Number(p.get('offset')||0);
 if(!Number.isSafeInteger(offset)||offset<0)bad('Invalid page offset.');
 const dailyArgs=[],dailyWhere=[];
 if(start){dailyWhere.push('day>=?');dailyArgs.push(start);}
 if(end){dailyWhere.push('day<=?');dailyArgs.push(end);}
 const dated=!!(start||end);
 // Lifetime totals include usage collected before daily/mode recording was introduced.
 const cte=`WITH d AS (SELECT person_id,SUM(active_seconds) total,
 SUM(CASE WHEN mode='vfr' THEN active_seconds ELSE 0 END) vfr_seconds,
 SUM(CASE WHEN mode='ifr' THEN active_seconds ELSE 0 END) ifr_seconds,
 SUM(CASE WHEN mode='unknown' THEN active_seconds ELSE 0 END) unknown_seconds
 FROM usage_daily ${dailyWhere.length?'WHERE '+dailyWhere.join(' AND '):''} GROUP BY person_id),
 metrics AS (SELECT p.id,p.display_name,p.kind,p.country,p.region,p.city,p.created_at,p.last_seen,
 ${dated?'COALESCE(d.total,0)':'p.active_seconds'} active_seconds,
 COALESCE(d.vfr_seconds,0) vfr_seconds,COALESCE(d.ifr_seconds,0) ifr_seconds,
 ${dated?'COALESCE(d.unknown_seconds,0)':'MAX(0,p.active_seconds-COALESCE(d.vfr_seconds,0)-COALESCE(d.ifr_seconds,0))'} unknown_seconds
 FROM people p LEFT JOIN d ON d.person_id=p.id),
 selected AS (SELECT *,${mode?mode+'_seconds':'active_seconds'} filtered_seconds FROM metrics)`;
 const args=[...dailyArgs],where=[];
 if(dated||mode)where.push('filtered_seconds>0');
 for(const [key,column] of [['search','display_name'],['country','country'],['region','region'],['city','city']]){
  const value=(p.get(key)||'').trim();if(value.length>100)bad('Filter text is too long.');
  if(value){where.push(`instr(lower(${column}),lower(?))>0`);args.push(value);}
 }
 if(kind){where.push('kind=?');args.push(kind);}
 if(p.get('excludeSelf')==='1'){where.push('id<>?');args.push(ownerId);}
 const filter=where.length?' WHERE '+where.join(' AND '):'';
 const sorts={hours:'filtered_seconds DESC,id',recent:'last_seen DESC,id',name:'display_name COLLATE NOCASE,id'};
 const sort=p.get('sort')||'recent';if(!sorts[sort])bad('Invalid sort order.');
 const rows=await env.DB.prepare(cte+' SELECT * FROM selected'+filter+' ORDER BY '+sorts[sort]+' LIMIT 101 OFFSET ?').bind(...args,offset).all();
 const summary=await env.DB.prepare(cte+` SELECT COUNT(*) people,COALESCE(SUM(kind='guest'),0) guests,COALESCE(SUM(kind='member'),0) members,
 COALESCE(SUM(filtered_seconds),0) active_seconds,
 COALESCE(SUM(${mode&&mode!=='vfr'?'0':'vfr_seconds'}),0) vfr_seconds,
 COALESCE(SUM(${mode&&mode!=='ifr'?'0':'ifr_seconds'}),0) ifr_seconds,
 COALESCE(SUM(${mode&&mode!=='unknown'?'0':'unknown_seconds'}),0) unknown_seconds FROM selected`+filter).bind(...args).first();
 const meta=await env.DB.prepare("SELECT value FROM usage_metadata WHERE key='daily_since'").first();
 return {people:rows.results.slice(0,100).map(row=>({...row,active_seconds:row.filtered_seconds,
 vfr_seconds:mode&&mode!=='vfr'?0:row.vfr_seconds,ifr_seconds:mode&&mode!=='ifr'?0:row.ifr_seconds,unknown_seconds:mode&&mode!=='unknown'?0:row.unknown_seconds})),summary,
 nextOffset:rows.results.length>100?offset+100:null,dailySince:meta?.value||null,timezone:'UTC'};
}

const enc=new TextEncoder();
export const digest=async text=>Array.from(new Uint8Array(await crypto.subtle.digest('SHA-256',enc.encode(text))),b=>b.toString(16).padStart(2,'0')).join('');
const secret=()=>Array.from(crypto.getRandomValues(new Uint8Array(32)),b=>b.toString(16).padStart(2,'0')).join('');
const clean=(v,n=100)=>typeof v==='string'?v.replace(/[\u0000-\u001f]/g,'').slice(0,n):'';
const uuid=v=>typeof v==='string'&&/^[0-9a-f-]{36}$/i.test(v);
const geo=req=>req.headers.get('X-FK-Usage')==='off'?{country:'',region:'',city:''}:({country:clean(req.cf?.country,2),region:clean(req.cf?.region,80),city:clean(req.cf?.city,80)});
const origins=env=>String(env.ALLOWED_ORIGINS||'').split(',').map(x=>x.trim()).filter(Boolean);
function response(data,status=200,origin=''){
 return new Response(JSON.stringify(data),{status,headers:{'Content-Type':'application/json','Cache-Control':'no-store','X-Content-Type-Options':'nosniff',...(origin?{'Access-Control-Allow-Origin':origin,'Vary':'Origin'}:{})}});
}
async function body(req){
 if(!req.headers.get('Content-Type')?.startsWith('application/json'))throw new Error('JSON required');
 const reader=req.body?.getReader();if(!reader)return {};
 let size=0,chunks=[];while(true){const {done,value}=await reader.read();if(done)break;size+=value.length;if(size>16000){await reader.cancel();throw Error('Request too large');}chunks.push(value);}
 const all=new Uint8Array(size);let at=0;for(const c of chunks){all.set(c,at);at+=c.length;}return JSON.parse(new TextDecoder().decode(all));
}
async function session(req,env){
 const token=req.headers.get('Authorization')?.match(/^Bearer ([0-9a-f]{64})$/)?.[1];if(!token)return null;
 const hash=await digest(token);
 const row=await env.DB.prepare('SELECT s.*,p.kind,p.display_name FROM sessions s JOIN people p ON p.id=s.person_id WHERE token_hash=? AND expires_at>?').bind(hash,Date.now()).first();
 // Remember sign-in for 30 days; require fresh Google verification for owner reports after one hour.
 return row?{...row,owner:row.owner && row.expires_at-30*86400000+3600000>Date.now()?1:0,hash}:null;
}
const profile=async(env,id)=>env.DB.prepare('SELECT id,display_name,kind,active_seconds,country,region,city FROM people WHERE id=?').bind(id).first();
export function createWorker(verify=verifyGoogleToken){return {
 async fetch(req,env){
  if(new URL(req.url).protocol!=='https:')return response({error:'HTTPS required'},400);
  const origin=req.headers.get('Origin')||'';
  if(!origins(env).includes(origin))return response({error:'Origin not allowed'},403);
  if(req.method==='OPTIONS')return new Response(null,{status:204,headers:{'Access-Control-Allow-Origin':origin,'Access-Control-Allow-Methods':'GET,POST,DELETE,OPTIONS','Access-Control-Allow-Headers':'Authorization,Content-Type,X-FK-Usage','Access-Control-Max-Age':'600','Vary':'Origin'}});
  if(!env.DB)return response({error:'Usage service not configured'},503,origin);
  try{
   if(!env.API_RATE_LIMITER)return response({error:'Rate limiter not configured'},503,origin);
   const rate=await env.API_RATE_LIMITER.limit({key:await digest(req.headers.get('CF-Connecting-IP')||'unknown')});
   if(!rate.success)return response({error:'Too many requests'},429,origin);
   const path=new URL(req.url).pathname;
   if(path==='/session'&&req.method==='POST'){
    const data=await body(req);if(Object.keys(data).length)return response({error:'Unexpected fields'},400,origin);
    const id=crypto.randomUUID(),token=secret(),hash=await digest(token),now=Date.now(),g=geo(req);
    await env.DB.batch([
     env.DB.prepare('INSERT INTO people(id,display_name,kind,country,region,city,created_at,last_seen) VALUES(?,?,?,?,?,?,?,?)').bind(id,'Guest '+id.slice(0,8),'guest',g.country,g.region,g.city,now,now),
     env.DB.prepare('INSERT INTO sessions(token_hash,person_id,expires_at,last_credit_ms) VALUES(?,?,?,?)').bind(hash,id,now+(req.headers.get('X-FK-Usage')==='off'?300000:90*86400000),now)
    ]);
    return response({token,profile:await profile(env,id),owner:false},201,origin);
   }
   const s=await session(req,env);if(!s)return response({error:'Sign in/session required'},401,origin);
   if(path==='/me'&&req.method==='GET')return response({profile:await profile(env,s.person_id),owner:!!s.owner},200,origin);
   if(path==='/me'&&req.method==='DELETE'){
    await env.DB.prepare('DELETE FROM people WHERE id=?').bind(s.person_id).run();return response({deleted:true},200,origin);
   }
   if(path==='/logout'&&req.method==='POST'){
    await env.DB.prepare('DELETE FROM sessions WHERE token_hash=?').bind(s.hash).run();return response({signedOut:true},200,origin);
   }
   if(path==='/challenge'&&req.method==='POST'){
    if(!env.GOOGLE_CLIENT_ID)return response({error:'Google sign-in not configured'},503,origin);
    const nonce=secret();await env.DB.prepare('UPDATE sessions SET nonce_hash=?,nonce_expires=? WHERE token_hash=?').bind(await digest(nonce),Date.now()+300000,s.hash).run();
    return response({nonce},200,origin);
   }
   if(path==='/login'&&req.method==='POST'){
    const data=await body(req);
    if(Object.keys(data).some(k=>!['credential','nonce'].includes(k))||typeof data.nonce!=='string'||data.nonce.length!==64||!s.nonce_hash||s.nonce_expires<Date.now()||await digest(data.nonce)!==s.nonce_hash)return response({error:'Invalid or expired sign-in challenge'},401,origin);
    // Consume the challenge once, atomically, before processing identity.
    const consumed=await env.DB.prepare('UPDATE sessions SET nonce_hash=NULL,nonce_expires=NULL WHERE token_hash=? AND nonce_hash=?').bind(s.hash,s.nonce_hash).run();
    if(!consumed.meta.changes)return response({error:'Challenge already used'},401,origin);
    let claims;try{claims=await verify(data.credential,env.GOOGLE_CLIENT_ID,data.nonce);}catch{return response({error:'Google identity could not be verified'},401,origin);}
    const sub=await digest('google:'+claims.sub),name=clean(claims.name)||'Signed-in pilot';
    const owner=claims.email_verified===true && String(claims.email||'').toLowerCase().endsWith('@gmail.com') &&
      String(env.OWNER_EMAILS||'').toLowerCase().split(',').map(x=>x.trim()).filter(Boolean).includes(String(claims.email).toLowerCase());
    const member=await env.DB.prepare('SELECT id FROM people WHERE google_sub=?').bind(sub).first();
    const now=Date.now(),token=secret(),hash=await digest(token),g=geo(req);
    let id=member?.id||s.person_id;
    const statements=[];
    if(!member){
     // Only a guest can become a new member. Switching members requires sign-out first.
     if(s.kind!=='guest')return response({error:'Sign out before switching accounts'},409,origin);
     statements.push(env.DB.prepare('UPDATE people SET google_sub=?,display_name=?,kind=\'member\' WHERE id=?').bind(sub,name,id));
    }else if(s.person_id!==id){
     if(s.kind!=='guest')return response({error:'Sign out before switching accounts'},409,origin);
     statements.push(env.DB.prepare('UPDATE people SET active_seconds=active_seconds+(SELECT active_seconds FROM people WHERE id=?) WHERE id=?').bind(s.person_id,id));
     statements.push(env.DB.prepare('INSERT INTO usage_daily(person_id,day,mode,active_seconds) SELECT ?,day,mode,active_seconds FROM usage_daily WHERE person_id=? ON CONFLICT(person_id,day,mode) DO UPDATE SET active_seconds=active_seconds+excluded.active_seconds').bind(id,s.person_id));
     statements.push(env.DB.prepare('DELETE FROM people WHERE id=?').bind(s.person_id));
    }
    if(req.headers.get('X-FK-Usage')==='off')statements.push(env.DB.prepare('UPDATE people SET display_name=?,last_seen=? WHERE id=?').bind(name,now,id));
    else statements.push(env.DB.prepare('UPDATE people SET display_name=?,country=?,region=?,city=?,last_seen=? WHERE id=?').bind(name,g.country,g.region,g.city,now,id));
    statements.push(env.DB.prepare('DELETE FROM sessions WHERE token_hash=?').bind(s.hash));
    statements.push(env.DB.prepare('INSERT INTO sessions(token_hash,person_id,expires_at,last_credit_ms,owner) VALUES(?,?,?,?,?)').bind(hash,id,now+30*86400000,now,owner?1:0));
    await env.DB.batch(statements);
    return response({token,profile:await profile(env,id),owner},200,origin);
   }
   if(path==='/usage'&&req.method==='POST'){
    if(req.headers.get('X-FK-Usage')==='off')return response({error:'Usage tracking is off'},403,origin);
    const data=await body(req);
    if(Object.keys(data).some(k=>!['id','seconds','mode'].includes(k))||!uuid(data.id)||(data.mode!==undefined&&!['vfr','ifr','unknown'].includes(data.mode))||!Number.isInteger(data.seconds)||data.seconds<1||data.seconds>90)return response({error:'Invalid usage batch'},400,origin);
    const now=Date.now(),g=geo(req);
    // An atomic insert/trigger deduplicates retries and caps credit to server elapsed time.
    await env.DB.batch([
     env.DB.prepare('INSERT OR IGNORE INTO usage_batches(id,person_id,session_hash,seconds,received_at,mode) SELECT ?,person_id,token_hash,MIN(?,MAX(0,CAST((?-last_credit_ms)/1000 AS INTEGER))),?,? FROM sessions WHERE token_hash=? AND expires_at>?').bind(data.id,data.seconds,now,now,data.mode||'unknown',s.hash,now),
     env.DB.prepare('UPDATE people SET country=?,region=?,city=? WHERE id=?').bind(g.country,g.region,g.city,s.person_id)
    ]);
    return response({accepted:true},200,origin);
   }
   if(path==='/admin/people'&&req.method==='GET'){
    if(!s.owner)return response({error:'Owner access only'},403,origin);
    return response(await ownerPeople(env,new URL(req.url).searchParams,s.person_id),200,origin);
   }
   return response({error:'Not found'},404,origin);
  }catch(e){return response({error:e.status===400?e.message:'Usage service temporarily unavailable'},e.status===400?400:503,origin);}
 },
 async scheduled(_event,env){
  const now=Date.now();await env.DB.batch([
   env.DB.prepare('DELETE FROM sessions WHERE expires_at<?').bind(now),
   env.DB.prepare('DELETE FROM usage_batches WHERE received_at<?').bind(now-2*86400000),
   env.DB.prepare('DELETE FROM people WHERE kind=\'guest\' AND last_seen<?').bind(now-180*86400000)
  ]);
 }
};}
export default createWorker();
